# Q31：手写发布订阅模式（EventEmitter）

```javascript
class EventEmitter {
  constructor() {
    this.events = {};
  }
  on(type, handler) {
    (this.events[type] || (this.events[type] = [])).push(handler);
  }
  off(type, handler) {
    if (!this.events[type]) return;
    this.events[type] = this.events[type].filter(h => h !== handler && h.listener !== handler);
  }
  once(type, handler) {
    const wrap = (...args) => {
      handler.apply(this, args);
      this.off(type, wrap);
    };
    wrap.listener = handler;
    this.on(type, wrap);
  }
  emit(type, ...args) {
    (this.events[type] || []).forEach(handler => handler.apply(this, args));
  }
}
```
