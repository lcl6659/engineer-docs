# Q33：并发请求控制（限制同时请求数量）

```javascript
function limitConcurrency(urls, max, callback) {
  const results = [];
  let index = 0; // 当前待请求的url索引
  let inProgress = 0;

  function request(url, idx) {
    inProgress++;
    fetch(url).then(res => res.json()).then(data => {
      results[idx] = data;
      inProgress--;
      while (index < urls.length && inProgress < max) {
        request(urls[index], index);
        index++;
      }
      if (inProgress === 0 && index >= urls.length) {
        callback(results);
      }
    }).catch(e => {
      results[idx] = e;
      inProgress--;
      while (index < urls.length && inProgress < max) {
        request(urls[index], index);
        index++;
      }
      if (inProgress === 0 && index >= urls.length) {
        callback(results);
      }
    });
  }
  // 初始化，启动前max个任务
  while (index < urls.length && inProgress < max) {
    request(urls[index], index);
    index++;
  }
}
```
